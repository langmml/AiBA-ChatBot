/*!
 * Empire ABA chat widget — single-file embed.
 * ---------------------------------------------------------------------------
 * CANONICAL SOURCE. This file is the only copy of the widget. The Worker
 * imports it as a string (see wrangler.toml [[rules]] type = "Text") and
 * serves it at GET /widget.js. Edit it here — nowhere else. Do not copy it
 * into a page or another folder; re-deploy the Worker instead.
 * ---------------------------------------------------------------------------
 * Drop ONE <script> tag on the site; this injects a floating chat launcher.
 * Everything renders inside a Shadow DOM, so it can't affect the host site's
 * styles and the site can't affect it.
 *
 * Config via data-attributes on the <script> tag (or window.EMPIRE_CHAT):
 *   data-endpoint (required)  Backend base URL — the Worker's origin, or a
 *                             full webhook URL when data-backend="make".
 *   data-mode   "ai" | "capture"   default "ai"
 *       "ai"      => free-text questions answered by Claude via <endpoint>/chat,
 *                    finished lead POSTed to <endpoint>/lead
 *       "capture" => no AI; scripted + canned FAQ only; finished lead POSTed to
 *                    <endpoint> (use with a plain webhook backend)
 *   data-agent  header name (default "Riley")
 *   data-title  header subtitle (default "Empire ABA")
 *   data-accent accent color (default "#2eaae1", Empire's blue)
 *   data-avatar URL to an agent photo (optional)
 * ---------------------------------------------------------------------------
 */
(function () {
  "use strict";

  var script = document.currentScript;
  var cfg = window.EMPIRE_CHAT || {};
  function attr(n, d) {
    // Accept data-chat-endpoint on the tag, and either "chat-endpoint" or
    // "chatEndpoint" in window.EMPIRE_CHAT.
    var camel = n.replace(/-([a-z])/g, function (_, c) { return c.toUpperCase(); });
    var v = (script && script.getAttribute("data-" + n)) || cfg[n] || cfg[camel];
    return v || d;
  }

  var ENDPOINT = (attr("endpoint", "") || "").replace(/\/$/, "");
  var MODE = attr("mode", "ai");
  // "worker" = one base URL, we append /chat and /lead.
  // "make"   = two separate webhook URLs, and POST as text/plain so the browser
  //            skips the CORS preflight (webhook backends rarely answer OPTIONS).
  var BACKEND = attr("backend", "worker");
  var DEBUG = String(attr("debug", "")) === "true";
  var AGENT = attr("agent", "Riley");
  var TITLE = attr("title", "Empire ABA");
  var ACCENT = attr("accent", "#2eaae1");
  var AVATAR = attr("avatar", "");

  var CHAT_URL = attr("chat-endpoint", "") || (ENDPOINT ? ENDPOINT + "/chat" : "");
  var LEAD_URL =
    attr("lead-endpoint", "") ||
    (ENDPOINT ? (MODE === "capture" || BACKEND === "make" ? ENDPOINT : ENDPOINT + "/lead") : "");
  var SIMPLE_POST = BACKEND === "make" || MODE === "capture";

  if (!LEAD_URL && !CHAT_URL) {
    console.warn("[Empire chat] No endpoint configured; widget disabled.");
    return;
  }
  if (DEBUG) console.log("[Empire chat] mode=" + MODE + " backend=" + BACKEND + " chat=" + (CHAT_URL || "(none)") + " lead=" + (LEAD_URL || "(none)"));

  var PHONES = {
    Wilmington: "(910) 773-1234",
    Triangle: "(984) 224-0007",
    "Charlotte Metro": "(704) 981-2343",
  };

  /* ----------------------------- state --------------------------------- */
  var lead = { interest: "", region: "", childAge: "", insurance: "", name: "", phone: "", email: "", notes: "" };
  var transcript = [];
  var step = "greeting";
  var sending = false;
  var opened = false;

  /* --------------------------- canned FAQ ------------------------------ */
  var FAQ = {
    "What is ABA?":
      "ABA helps your child master real-life skills \u2014 communication, social, daily living, and safety \u2014 through the science of how people learn. We study what motivates your child, set goals with your family, and build the skills that matter most to you.",
    "Do you take my insurance?":
      "We accept Medicaid and most major commercial plans (Blue Cross NC, UnitedHealthcare, Aetna, Cigna) plus NC Medicaid plans like Alliance, Vaya, Trillium, Partners, Healthy Blue, WellCare, Carolina Complete, and AmeriHealth Caritas. We verify your exact benefits for free before anything starts.",
    "What's included?":
      "A full package: in-home therapy, parent support, school collaboration, and community-based support \u2014 so progress carries across every setting.",
    "Do you come to us?":
      "Yes \u2014 we provide in-home ABA across Wilmington, the Triangle (Raleigh\u2013Durham), and Charlotte Metro.",
    "How many hours per week?":
      "It varies by child \u2014 a BCBA sets the right amount after an assessment, woven into normal daily routines. We'll walk you through what makes sense for your child.",
    "How do we start?":
      "Just your name, number, and email \u2014 our team reaches out, verifies insurance, and guides you through every step.",
  };

  /* --------------------------- scripted flow --------------------------- */
  var FLOW = {
    greeting: {
      say: ["Hi! \uD83D\uDC4B I'm " + AGENT + " with Empire ABA.", "Are you looking into ABA therapy for your child?"],
      chips: ["Yes, for my child", "I have questions", "I'm interested in a job"],
      handle: function (t) {
        lead.interest = t;
        if (/job|career|hiring|work/i.test(t)) return jobPath();
        if (/question/i.test(t)) return faqMenu();
        return runStep("region");
      },
    },
    region: {
      say: ["What city or area are you in?"],
      chips: ["Wilmington", "Triangle / Raleigh-Durham", "Charlotte Metro", "Another area"],
      handle: function (t) {
        if (/wilmington/i.test(t)) lead.region = "Wilmington";
        else if (/triangle|raleigh|durham/i.test(t)) lead.region = "Triangle";
        else if (/charlotte/i.test(t)) lead.region = "Charlotte Metro";
        else lead.region = "Other";
        if (lead.region === "Other") {
          botThen(
            "Great \u2014 we serve that area.",
            function () { runStep("age"); }
          );
          return;
        }
        runStep("age");
      },
    },
    age: {
      say: ["How old is your child?"],
      chips: ["Under 3", "3\u20135", "6\u201310", "11+"],
      handle: function (t) { lead.childAge = t; runStep("insurance"); },
    },
    insurance: {
      say: ["Thanks! Which insurance do you have?"],
      chips: ["Medicaid", "Blue Cross NC", "UnitedHealthcare", "Aetna", "Cigna", "Other / not sure"],
      handle: function (t) {
        lead.insurance = t;
        botSeq(
          [
            "Perfect \u2014 we accept Medicaid and most major commercial plans, and our team verifies your exact benefits for free before anything starts.",
            "Let's get you set up so someone can reach out. What's your name?",
          ],
          function () { step = "name"; }
        );
      },
    },
    name: {
      handle: function (t) { lead.name = t; askPhone(); },
    },
    phone: {
      validate: function (t) { return (t.match(/\d/g) || []).length >= 7 ? null : "Hmm, that looks short \u2014 mind re-entering the number?"; },
      handle: function (t) { lead.phone = t; botThen("And your email?", function () { step = "email"; }); },
    },
    email: {
      validate: function (t) { return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(t.trim()) ? null : "That email looks off \u2014 mind checking it?"; },
      handle: function (t) {
        lead.email = t.trim();
        botThen("Almost done! Anything you'd like the team to know? (Or tap Skip.)", function () { step = "notes"; showChips(["Skip"]); });
      },
    },
    notes: {
      handle: function (t) { if (!/^skip$/i.test(t)) lead.notes = t; finish(); },
    },
  };

  function askPhone() {
    var n = firstName(lead.name);
    botThen("Nice to meet you" + (n ? ", " + n : "") + "! What's the best phone number to reach you?", function () { step = "phone"; });
  }

  function jobPath() {
    lead.interest = "Job seeker";
    botSeq(
      ["Love that! You can see openings at empireaba.com/enrollment.", "Want someone to reach out? What's your name?"],
      function () { step = "name"; }
    );
  }

  function faqMenu() {
    step = "faq";
    botThen("Sure! What would you like to know?", function () {
      showChips(Object.keys(FAQ).concat(["Actually, let's get started"]));
    });
  }

  /* --------------------------- flow engine ----------------------------- */
  function runStep(name) {
    step = name;
    var s = FLOW[name];
    if (!s || !s.say) return;
    var lines = typeof s.say === "function" ? s.say() : s.say;
    botSeq(lines, function () { if (s.chips) showChips(s.chips); });
  }

  function submit(text) {
    text = (text || "").trim();
    if (!text || sending) return;
    input.value = "";
    clearChips();
    addMe(text);
    harvestContact(text);
    scheduleInactiveLead();
    route(text);
  }

  function route(text) {
    // Full AI mode: Claude handles everything, including gathering details.
    if (MODE === "ai") return answerWithAI(text);

    // FAQ menu selections
    if (step === "faq") {
      // In AI mode we let Claude answer even the listed questions, so the bot
      // sounds live. Canned answers are for capture mode (no AI available).
      if (FAQ[text] && MODE !== "ai") {
        botThen(FAQ[text], function () {
          botThen("Want us to reach out and get you set up?", function () {
            showChips(["Yes, let's go", "Ask another question"]);
            step = "faq_followup";
          });
        });
        return;
      }
      if (/started|go|yes/i.test(text)) return runStep("region");
      // typed free text in FAQ menu
      return answerFree(text, function () { faqMenu(); });
    }
    if (step === "faq_followup") {
      if (/another|question/i.test(text)) return faqMenu();
      return runStep("region");
    }

    var s = FLOW[step];

    // If they ask a question at any point, answer it, then resume the flow.
    // (Chip labels never look like questions, so taps still follow the script.)
    if (s && looksLikeQuestion(text)) {
      answerFree(text, function () { reAsk(step); });
      return;
    }

    if (!s || !s.handle) {
      // after finish: free chat (AI) or gentle capture (non-AI)
      return answerFree(text);
    }

    if (s.validate) {
      var err = s.validate(text);
      if (err) return botThen(err);
    }
    s.handle(text);
  }

  function reAsk(name) {
    var s = FLOW[name];
    if (!s) return;
    if (name === "phone") return askPhone();
    if (name === "email") return botThen("And your email?");
    if (name === "name") return botThen("What's your name?");
    if (!s.say) return;
    var lines = typeof s.say === "function" ? s.say() : s.say;
    botThen(lines[lines.length - 1], function () { if (s.chips) showChips(s.chips); });
  }

  // Free-text handler: AI mode -> Claude; capture mode -> keyword FAQ or catch-all.
  function answerFree(text, then) {
    if (MODE === "ai") return answerWithAI(text, then);
    var hit = keywordFaq(text);
    if (hit) {
      botThen(FAQ[hit], function () { if (then) then(); else botThen("Want us to reach out? What's the best number to reach you?", function () { step = "phone"; }); });
      return;
    }
    botThen("Great question \u2014 our team can go deep on that for you. What's the best number to reach you?", function () {
      step = "phone";
      if (then) then();
    });
  }

  function keywordFaq(t) {
    t = t.toLowerCase();
    if (/insur|medicaid|cover|plan|blue cross|aetna|cigna|united/.test(t)) return "Do you take my insurance?";
    if (/hour|how (much|long|often)|week|schedule/.test(t)) return "How many hours per week?";
    if (/where|area|come|home|location|serve/.test(t)) return "Do you come to us?";
    if (/what is aba|what.?s aba|how.*work|about aba/.test(t)) return "What is ABA?";
    if (/start|begin|sign up|enroll|first step/.test(t)) return "How do we start?";
    if (/includ|offer|service/.test(t)) return "What's included?";
    return null;
  }

  /* ------------------------------ AI ----------------------------------- */
  function answerWithAI(text, then) {
    sending = true;
    showTyping();
    fetch(CHAT_URL, buildPost({
      transcriptText: transcriptAsText(),
      lastMessage: text || "",
      page: location.href,
    }, { messages: aiMessages(), transcriptText: transcriptAsText(), lead: lead, page: location.href }))
      // Read as text first. A webhook endpoint may return JSON, plain text, or
      // the word "Accepted" (which means it never sent a reply back).
      .then(function (r) {
        return r.text().then(function (t) { return { ok: r.ok, status: r.status, text: t }; });
      })
      .then(function (res) {
        hideTyping();
        if (DEBUG) console.log("[Empire chat] HTTP " + res.status + " <- " + res.text.slice(0, 400));

        if (!res.ok) return problem("HTTP " + res.status + " from the backend. " + hint(res.status));

        var raw = (res.text || "").trim();
        if (!raw) return problem("Backend returned an empty body.");

        // "Accepted" = the endpoint took the request but returned no reply.
        if (/^accepted$/i.test(raw)) {
          return problem('Backend replied "Accepted" with no content \u2014 it needs to return the reply text in the response body.');
        }

        var reply = "";
        try {
          var d = JSON.parse(raw);
          reply = d.reply || d.text || d.content || "";
          if (!reply) return problem("Response JSON had no reply field. Got: " + raw.slice(0, 200));
        } catch (e) {
          reply = raw; // plain-text reply is fine too
        }

        var m = reply.match(/\[\[(LEAD|DONE)([^\]]*)\]\]/i);
        if (m) {
          reply = reply.replace(m[0], "").trim();
          parseLeadFields(m[2] || "");
        }
        addBot(reply);
        // A completed conversation sends immediately. Partial contact details
        // wait until the visitor has been inactive for a full minute.
        if (m && /^DONE$/i.test(m[1] || "")) completeLead();
        else scheduleInactiveLead();
      })
      .catch(function (err) {
        hideTyping();
        if (DEBUG) console.error("[Empire chat] request failed:", err);
        problem("Could not reach the backend (network or CORS). Check the console, and make sure the response sends an Access-Control-Allow-Origin header.");
      })
      .finally(function () { sending = false; if (then) setTimeout(then, 400); });
  }

  // Builds the fetch options. For plain webhook backends we send
  // application/x-www-form-urlencoded: it's a CORS "simple request" (no
  // preflight) AND most of them parse it into fields natively, so no
  // JSON-parsing step is needed. The Worker gets normal JSON.
  function buildPost(flatFields, jsonBody) {
    if (!SIMPLE_POST) {
      return { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify(jsonBody) };
    }
    if (PAYLOAD === "form") {
      return {
        method: "POST",
        headers: { "content-type": "application/x-www-form-urlencoded;charset=UTF-8" },
        body: formEncode(flatFields),
      };
    }
    // Default for webhook backends: JSON body sent as text/plain. Still a CORS
    // "simple request" (no preflight), and it's the shape most of them expect.
    return { method: "POST", headers: { "content-type": "text/plain;charset=UTF-8" }, body: JSON.stringify(jsonBody) };
  }

  function formEncode(obj) {
    var parts = [];
    for (var k in obj) {
      if (!Object.prototype.hasOwnProperty.call(obj, k)) continue;
      var v = obj[k];
      if (v === undefined || v === null) v = "";
      parts.push(encodeURIComponent(k) + "=" + encodeURIComponent(String(v)));
    }
    return parts.join("&");
  }

  // "json" (default, for backends that parse a JSON body) or "form"
  var PAYLOAD = String(attr("payload", "json")).toLowerCase();
  var leadSentSig = "";
  var LEAD_IDLE_MS = 60 * 1000;
  var leadIdleTimer = null;
  var leadLastActivityAt = 0;

  function leadSignature() {
    return [lead.name, lead.phone, lead.email, lead.region, lead.childAge, lead.insurance].join("|");
  }

  // Backup extraction: if the visitor types an email or phone, capture it
  // ourselves. This means a lead survives even if the AI forgets the marker.
  function harvestContact(t) {
    var em = t.match(/[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}/i);
    if (em && !lead.email) lead.email = em[0];
    if (!lead.phone) {
      var digits = (t.match(/\d/g) || []).length;
      if (digits >= 10 && digits <= 11 && /^[\d\s().+-]+$/.test(t.trim())) lead.phone = t.trim();
    }
  }

  function clearLeadIdleTimer() {
    if (!leadIdleTimer) return;
    clearTimeout(leadIdleTimer);
    leadIdleTimer = null;
  }

  // Wait for one full minute with no new visitor activity before sending a
  // partial lead. Each new message resets this timer.
  function scheduleInactiveLead() {
    clearLeadIdleTimer();
    if (!LEAD_URL || (!lead.phone && !lead.email)) return;
    if (leadSignature() === leadSentSig) return;
    leadLastActivityAt = Date.now();
    leadIdleTimer = setTimeout(function () {
      leadIdleTimer = null;
      sendLeadIfChanged("inactive");
    }, LEAD_IDLE_MS);
  }

  function sendLeadIfChanged(status) {
    if (!LEAD_URL) return;
    if (!lead.phone && !lead.email) return; // nothing to act on yet
    var sig = leadSignature();
    if (sig === leadSentSig) return; // nothing new since last send
    leadSentSig = sig;
    sendLead(status || "partial");
  }

  function completeLead() {
    clearLeadIdleTimer();
    sendLeadIfChanged("complete");
  }

  // If a browser throttled the timer while the page was hidden, send on page
  // teardown only when the full inactivity period has already elapsed.
  function flushLead() {
    if (!LEAD_URL) return;
    if (!lead.phone && !lead.email) return;
    if (!leadLastActivityAt || Date.now() - leadLastActivityAt < LEAD_IDLE_MS) return;
    if (leadSignature() === leadSentSig) return;
    leadSentSig = leadSignature();
    try {
      var payload = JSON.stringify(leadPayload("abandoned"));
      if (navigator.sendBeacon) {
        navigator.sendBeacon(LEAD_URL, new Blob([payload], { type: "text/plain;charset=UTF-8" }));
      }
    } catch (e) {
      /* nothing more we can do at unload */
    }
  }

  function hint(status) {
    if (status === 400) return "The endpoint rejected the request \u2014 check that it's switched on and accepting POSTs.";
    if (status === 404 || status === 410) return "That endpoint URL wasn't found \u2014 double-check it.";
    if (status >= 500) return "The endpoint errored \u2014 check its logs.";
    return "";
  }

  // Visitors can see whatever lands in the chat, so the bubble always stays
  // warm and non-technical. The real reason goes to the console for debugging.
  function problem(msg) {
    if (DEBUG) console.warn("[Empire chat] " + msg);
    addBot("Sorry \u2014 I'm having a little trouble on my end right now. Leave your name and the best number to reach you, and the team will follow up soon.");
  }

  // Pulls key="value" pairs out of the [[DONE ...]] marker, if Claude sent any.
  // The lead email always includes the full transcript too, so this is a bonus,
  // not something the flow depends on.
  function parseLeadFields(s) {
    var re = /(\w+)\s*=\s*"([^"]*)"/g, m;
    while ((m = re.exec(s))) {
      var k = m[1].toLowerCase();
      var v = m[2];
      if (k === "name") lead.name = v;
      else if (k === "phone") lead.phone = v;
      else if (k === "email") lead.email = v;
      else if (k === "region") lead.region = v;
      else if (k === "age" || k === "childage") lead.childAge = v;
      else if (k === "insurance") lead.insurance = v;
      else if (k === "notes") lead.notes = v;
    }
  }

  function aiMessages() {
    return transcript
      .filter(function (m) { return m.role === "user" || m.role === "assistant"; })
      .slice(-12)
      .map(function (m) { return { role: m.role === "user" ? "user" : "assistant", content: m.content }; });
  }

  // A ready-to-read plain-text version of the conversation. Useful for
  // webhook backends that take one big text prompt — you can map this string
  // straight in, no JSON handling needed.
  function transcriptAsText() {
    return transcript
      .filter(function (m) { return m.role === "user" || m.role === "assistant"; })
      .slice(-12)
      .map(function (m) { return (m.role === "user" ? "Visitor: " : "Riley: ") + m.content; })
      .join("\n");
  }
  function finish() {
    var n = firstName(lead.name);
    var num = PHONES[lead.region];
    var tail = num ? " If you'd rather call now, we're at " + num + "." : "";
    botThen("Thank you" + (n ? ", " + n : "") + "! \uD83C\uDF89 Someone from our Empire ABA team will reach out very soon." + tail + " Talk soon!", function () {
      completeLead();
    });
  }
  function leadPayload(status) {
    return {
      lead: lead,
      status: status || "partial",
      transcript: transcript,
      transcriptText: transcriptAsText(),
      source: location.href,
    };
  }

  function sendLead(status) {
    if (!LEAD_URL) { if (DEBUG) console.warn("[Empire chat] no lead endpoint set; skipping"); return; }
    var flat = {
      name: lead.name, phone: lead.phone, email: lead.email,
      region: lead.region, childAge: lead.childAge, insurance: lead.insurance,
      interest: lead.interest, notes: lead.notes, status: status || "partial",
      transcriptText: transcriptAsText(), source: location.href,
    };
    if (DEBUG) console.log("[Empire chat] sending lead (" + (status || "partial") + ")", flat);
    fetch(LEAD_URL, buildPost(flat, leadPayload(status)))
      .then(function () { if (DEBUG) console.log("[Empire chat] lead sent"); })
      .catch(function (e) { console.warn("[Empire chat] lead POST failed", e); });
  }

  function openPanel() {
    opened = true; badge.style.display = "none"; launcher.style.display = "none";
    panel.classList.add("open"); input.focus();
    if (transcript.length) return;
    if (MODE === "ai") {

  /* --------------------------- rendering ------------------------------- */
  function botSeq(lines, done) { seq(lines, 0, done); }
  function seq(lines, i, done) {
    if (i >= lines.length) return done && done();
    showTyping();
    var delay = Math.min(950, 350 + lines[i].length * 11);
    setTimeout(function () { hideTyping(); addBot(lines[i]); seq(lines, i + 1, done); }, delay);
  }
  function botThen(line, done) { botSeq([line], done); }

  function addBot(text) {
    transcript.push({ role: "assistant", content: text });
    var row = document.createElement("div");
    row.className = "row bot";
    row.innerHTML = '<div class="mav">' + avImg + '</div><div class="bubble"></div>';
    row.querySelector(".bubble").textContent = text;
    body.appendChild(row); scroll();
  }
  function addMe(text) {
    transcript.push({ role: "user", content: text });
    var row = document.createElement("div");
    row.className = "row me";
    row.innerHTML = '<div class="bubble"></div>';
    row.querySelector(".bubble").textContent = text;
    body.appendChild(row); scroll();
  }
  function showChips(list) {
    var wrap = document.createElement("div");
    wrap.className = "chips";
    list.forEach(function (label) {
      var b = document.createElement("button");
      b.className = "chip"; b.textContent = label;
      b.addEventListener("click", function () { submit(label); });
      wrap.appendChild(b);
    });
    body.appendChild(wrap); scroll();
  }
  function clearChips() { body.querySelectorAll(".chips").forEach(function (c) { c.remove(); }); }

  var typingEl = null;
  function showTyping() {
    if (typingEl) return;

    row.className = "row bot";
    row.innerHTML = '<div class="mav">' + avImg + '</div><div class="typing"><span></span><span></span><span></span></div>';
    body.appendChild(row); typingEl = row; scroll();
  }
  function hideTyping() { if (typingEl) { typingEl.remove(); typingEl = null; } }
  function scroll() { body.scrollTop = body.scrollHeight; }
